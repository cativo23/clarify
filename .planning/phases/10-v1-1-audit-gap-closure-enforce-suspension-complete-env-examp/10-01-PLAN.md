       ```ts
       export const ACCOUNT_SUSPENDED = "ACCOUNT_SUSPENDED" as const;

       export async function assertNotSuspended(event: H3Event, userId?: string): Promise<void> {
         const client = await serverSupabaseClient(event);
         let uid = userId;
         if (!uid) {
           const { data } = await client.auth.getUser();
           if (!data.user) return; // unauth path handled elsewhere
           uid = data.user.id;
         }
         const { data, error } = await client
           .from("users")
           .select("is_suspended")
           .eq("id", uid)
           .maybeSingle();
         if (error) {
           console.error("[Auth] suspension lookup failed:", error.message);
           return; // fail-open on infra error matches existing admin_emails pattern
         }
         if (data?.is_suspended === true) {
           throw createError({
             statusCode: 403,
             statusMessage: "Account suspended",
             data: { code: ACCOUNT_SUSPENDED },
           });
         }
       }
       ```

## STRIDE Threat Register

| ID | Category | Component | Disposition | Mitigation |
|----|----------|-----------|-------------|------------|
| T-10-01 | Elevation of Privilege | suspended user calling /api/upload | mitigate | assertNotSuspended called after auth, before any work |
| T-10-02 | Elevation of Privilege | suspended user's pre-existing queued job | mitigate | Worker checks is_suspended at job pickup (not at enqueue), returns failed with no credit consumption |
| T-10-03 | Tampering | bypass via skipping helper on a new endpoint | accept | Documented in CLAUDE.md follow-up; this phase covers upload+worker; future endpoints must call assertNotSuspended (linted in code review) |
| T-10-04 | Information Disclosure | error message leaks internal state | mitigate | Error contract uses fixed code "ACCOUNT_SUSPENDED" — no user data, no DB error text |
| T-10-05 | Denial of Service | flapping suspension causes queue churn | accept | Admin action, low frequency, BullMQ retries already capped at 3 |
</threat_model>
